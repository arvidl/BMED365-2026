\section{Bioethics and AI}

\begin{frame}{A01: Discuss the four bioethical principles in an AI context}
    \textbf{\href{https://en.wikipedia.org/wiki/Principlism}{Beauchamp \& Childress' Four Principles}:}
    
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{1. Autonomy} (Respect for self-determination)
            \begin{itemize}
                \item Informed consent for AI use
                \item Right to know if AI is involved
                \item Ability to opt out of AI decisions
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{2. Beneficence} (Do good)
            \begin{itemize}
                \item AI should improve patient outcomes
                \item Validated and effective
                \item Accessible to all
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{3. Non-maleficence} (Primum non nocere)
            \begin{itemize}
                \item Avoid harm from incorrect predictions
                \item Minimize bias and discrimination
                \item Safety testing before clinical use
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{4. Justice} (Fair distribution)
            \begin{itemize}
                \item Equal access to AI-enhanced healthcare
                \item Avoid systematic discrimination
                \item Representative training data
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.3cm}
    \begin{block}{Challenge}
        The principles can conflict -- e.g. efficiency (beneficence) vs. privacy (autonomy)
    \end{block}
\end{frame}

\begin{frame}{A02: Identify types of bias in medical AI systems}
    \textbf{Bias = systematic skew leading to unfair outcomes}
    
    \vspace{0.3cm}
    \textbf{Sources of bias:}
    \begin{enumerate}
        \item \textbf{Historical bias:} Training data reflects historical unfairness
        \begin{itemize}
            \item Example: Underrepresentation of minorities in clinical studies
        \end{itemize}
        
        \item \textbf{Representation bias:} Skewed patient population in training data
        \begin{itemize}
            \item Example: Model trained only on white patients
        \end{itemize}
        
        \item \textbf{Measurement bias:} Incorrect proxy variables
        \begin{itemize}
            \item Example: Using healthcare costs as a proxy for disease severity (correlates with race)
        \end{itemize}
        
        \item \textbf{Evaluation bias:} Test data is not representative
        
        \item \textbf{Aggregation bias:} One model for all, ignoring subgroups
    \end{enumerate}
    
    \begin{alertblock}{Consequence}
        Bias can cause AI systems to provide worse healthcare to already marginalized groups.
    \end{alertblock}
\end{frame}

\begin{frame}{A03: Explain privacy considerations when using LLMs}
    \textbf{Privacy challenges with LLMs:}
    
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Data leakage to third parties:}
            \begin{itemize}
                \item Input is sent to LLM provider
                \item Potentially used for retraining
                \item Stored on foreign servers
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{Memorization:}
            \begin{itemize}
                \item LLMs may have ``learned'' personal data
                \item Can generate sensitive info in output
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Identification:}
            \begin{itemize}
                \item Re-identification from seemingly anonymous data
                \item Prompts can reveal context
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{Countermeasures:}
            \begin{itemize}
                \item Never share patient-identifiable info with public LLMs
                \item Use local/secure solutions (e.g. chat.uib.no)
                \item Anonymization and pseudonymization
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.3cm}
    \begin{alertblock}{Important rule}
        \textbf{Never} enter patient data into ChatGPT, Claude or other public LLMs!
    \end{alertblock}
\end{frame}

\section{EU AI Act}

\begin{frame}{A04: Describe the main features of the EU AI Act}
    \textbf{\href{https://artificialintelligenceact.eu/}{EU AI Act} (adopted 2024):}
    \begin{itemize}
        \item World's first comprehensive AI regulation
        \item Risk-based approach
        \item Applies to all who offer or use AI in the EU
    \end{itemize}
    
    \vspace{0.3cm}
    \textbf{Main principles:}
    \begin{enumerate}
        \item \textbf{Risk classification:} AI systems categorized by risk level
        \item \textbf{Requirements proportional to risk:} Higher risk = stricter requirements
        \item \textbf{Transparency:} Users must be informed about AI use
        \item \textbf{Human oversight:} Human-in-the-loop for high-risk systems
        \item \textbf{Documentation:} Technical documentation and logging
        \item \textbf{CE marking:} Conformity assessment before deployment
    \end{enumerate}
    
    \vspace{0.3cm}
    \begin{block}{Timeline}
        Gradual implementation 2024--2027. High-risk AI requirements apply from 2026.
    \end{block}
\end{frame}

\begin{frame}{A05: Explain risk classification in the EU AI Act}
    \textbf{Four risk levels:}
    
    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{Unacceptable risk} (PROHIBITED)
        \begin{itemize}
            \item Social scoring, manipulation, real-time facial recognition
        \end{itemize}
        
        \item \textbf{High risk} (STRICT REQUIREMENTS)
        \begin{itemize}
            \item Medical devices and diagnostics
            \item Critical infrastructure, education, labor market
            \item Requires risk assessment, documentation, human oversight
        \end{itemize}
        
        \item \textbf{Limited risk} (TRANSPARENCY REQUIREMENTS)
        \begin{itemize}
            \item Chatbots, deepfakes
            \item Must inform users that it is AI
        \end{itemize}
        
        \item \textbf{Minimal risk} (NO SPECIFIC REQUIREMENTS)
        \begin{itemize}
            \item Spam filters, game AI
            \item Voluntary ethical guidelines
        \end{itemize}
    \end{enumerate}
    
    \begin{alertblock}{Medical AI}
        Most medical AI systems will be classified as \textbf{high risk}.
    \end{alertblock}
\end{frame}

\begin{frame}{A06: Discuss requirements for high-risk AI in healthcare}
    \textbf{Requirements for high-risk AI systems:}
    
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Before deployment:}
            \begin{itemize}
                \item Risk assessment and management
                \item High data quality and representativeness
                \item Technical documentation
                \item Transparency and user guidance
                \item Cybersecurity
                \item Conformity assessment (CE marking)
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{During use:}
            \begin{itemize}
                \item Automatic logging
                \item Human oversight
                \item Post-market surveillance
                \item Incident reporting
                \item Periodic re-evaluation
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.3cm}
    \textbf{Specific to medical AI:}
    \begin{itemize}
        \item Coordinated with \href{https://eur-lex.europa.eu/legal-content/EN/TXT/?uri=CELEX:32017R0745}{MDR} (Medical Devices Regulation)
        \item Clinical evidence and validation required
        \item Requirements for explainability and robustness
    \end{itemize}
\end{frame}

\section{Responsibility and Fairness}

\begin{frame}{A07: Reflect on responsibility allocation when AI fails}
    \textbf{Who is responsible when AI makes mistakes?}
    
    \vspace{0.3cm}
    \textbf{Potential responsible parties:}
    \begin{itemize}
        \item \textbf{AI developer:} Design, training, validation
        \item \textbf{Healthcare institution:} Procurement, implementation, training
        \item \textbf{Clinician:} Uses AI, makes final decision
        \item \textbf{Patient:} Consents to AI use?
    \end{itemize}
    
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Key issues:}
            \begin{itemize}
                \item AI as ``black box'' -- difficult to understand errors
                \item Shared responsibility $\rightarrow$ diluted responsibility?
                \item Automation bias -- over-reliance on AI
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Legal developments:}
            \begin{itemize}
                \item AI Liability Directive (EU)
                \item Product liability extended to AI
                \item Burden of proof may be placed on AI provider
            \end{itemize}
        \end{column}
    \end{columns}
    
    \begin{block}{Core principle}
        The clinician retains ultimate responsibility -- AI is a \textbf{tool}, not a replacement.
    \end{block}
\end{frame}

\begin{frame}{A08: Know about GDPR-relevant aspects of AI}
    \textbf{\href{https://gdpr.eu/}{GDPR} and AI in healthcare:}
    
    \vspace{0.3cm}
    \textbf{Key principles:}
    \begin{itemize}
        \item \textbf{Lawful basis:} Consent, necessity, legitimate interest
        \item \textbf{Purpose limitation:} Data only for stated purpose
        \item \textbf{Data minimization:} Only necessary data
        \item \textbf{Storage limitation:} Do not store longer than necessary
    \end{itemize}
    
    \vspace{0.3cm}
    \textbf{Special provisions:}
    \begin{itemize}
        \item \textbf{Art. 22:} Right not to be subject to automated decisions
        \item \textbf{Art. 9:} Health data is a ``special category'' -- extra protected
        \item \textbf{Art. 35:} \href{https://gdpr.eu/data-protection-impact-assessment-template/}{DPIA} (Data Protection Impact Assessment) for high-risk processing
    \end{itemize}
    
    \vspace{0.3cm}
    \begin{alertblock}{Important}
        Patient data used for AI training/use requires a lawful basis and often a DPIA.
    \end{alertblock}
\end{frame}

\begin{frame}{A09: Discuss algorithmic fairness}
    \textbf{Fairness in ML: No universal definition}
    
    \vspace{0.3cm}
    \textbf{Different fairness definitions:}
    \begin{enumerate}
        \item \textbf{Demographic parity:} Equal proportion of positive predictions per group
        \item \textbf{Equalized odds:} Equal TPR and FPR per group
        \item \textbf{Calibration:} Confidence scores mean the same for all groups
        \item \textbf{Individual fairness:} Similar individuals are treated similarly
    \end{enumerate}
    
    \vspace{0.3cm}
    \textbf{The problem:}
    \begin{itemize}
        \item These definitions are often \textbf{mathematically incompatible}
        \item Choice of definition is a \textbf{value-laden} decision
        \item ``Fair'' in one sense can be ``unfair'' in another
    \end{itemize}
    
    \vspace{0.3cm}
    \begin{block}{Medical example}
        Should a risk model give the same score to a 30-year-old and a 70-year-old with the same symptoms? Age is a risk factor -- is that discrimination?
    \end{block}
\end{frame}
