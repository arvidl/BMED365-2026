\section{Trustworthy AI}

\begin{frame}{T01: Define trustworthy AI according to EU guidelines}
    \textbf{\href{https://digital-strategy.ec.europa.eu/en/policies/expert-group-ai}{EU High-Level Expert Group on AI} (2019):}
    
    \begin{block}{Definition}
        \textbf{Trustworthy AI} er AI that is lovlig, etisk og robust -- både teknisk og sosialt.
    \end{block}
    
    \vspace{0.2cm}
    \textbf{Syv nøkkelkrav:}
    \begin{enumerate}
        \item \textbf{Menneskers handlingsrom og tilsyn} -- Human agency \& oversight
        \item \textbf{Teknisk robustness og sikkerhet} -- Technical robustness \& safety
        \item \textbf{Privacy og dataforvaltning} -- Privacy \& data governance
        \item \textbf{Transparens} -- Transparency
        \item \textbf{Mangfold, ikke-diskriminering, fairness} -- Diversity, non-discrimination, fairness
        \item \textbf{Sosial og miljømessig velferd} -- Societal \& environmental well-being
        \item \textbf{Responsibilitylighet} -- Accountability
    \end{enumerate}
    
    \vspace{0.1cm}
    {\small
    \begin{alertblock}{Medical relevance}
        Høyrisiko AI-systemer i helse må oppfylle disse kravene under \href{https://artificialintelligenceact.eu/}{EU AI Act}.
    \end{alertblock}
    }
\end{frame}

\section{Robustness and Uncertainty}

\begin{frame}{T02: Explain the concept of robustness in ML/AI}
    \textbf{Robustness = modellens evne til å prestere pålitelig under variasjon}
    
    \vspace{0.2cm}
    \textbf{En robust modell:}
    \begin{itemize}
        \item Gir \textbf{konsistente resulter} på lignende input
        \item \textbf{Degraderer gradvis} (\href{https://en.wikipedia.org/wiki/Graceful_degradation}{ikke katastrofalt}) ved støy
        \item Generaliserer godt til nye, usette data
    \end{itemize}
    
    \vspace{0.2cm}
    \textbf{Typer robustness:}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item \textbf{Støyrobustness:} Toleranse for tilfeldig støy
                \item \textbf{Distribusjonell robustness:} Endringer i dataadvantageing
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item \textbf{Adversarial robustness:} Motstand mot bevisste angrep
                \item \textbf{Temporal robustness:} Stabilitet over tid
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.15cm}
    {\small
    \begin{block}{In medicine}
        En robust medical AI gir pålitelige predictioner uavhengig av variasjon i bildekvalitet, patientpopulasjon, or utstyr.
    \end{block}
    }
\end{frame}

\begin{frame}{T03: Describe data drift and its consequences}
    \textbf{\href{https://en.wikipedia.org/wiki/Data_drift}{Datadrift} (distributional shift):}
    \begin{itemize}
        \item Fotherwisekjell mellom \textbf{training data} og \textbf{produksjonsdata}
        \item Modellen møter data som ikke ligner det den er trent på
    \end{itemize}
    
    \vspace{0.2cm}
    \textbf{Typer drift:}
    \begin{enumerate}
        \item \textbf{\href{https://en.wikipedia.org/wiki/Covariate_shift}{Covariate shift}:} Input-advantageing endres (e.g. ny patientdemografi)
        \item \textbf{\href{https://en.wikipedia.org/wiki/Label_shift}{Label shift}:} Forekomst av klasser endres (e.g. pandemier)
        \item \textbf{\href{https://en.wikipedia.org/wiki/Concept_drift}{Concept drift}:} Sammenhengen mellom input og output endres
    \end{enumerate}
    
    \vspace{0.2cm}
    \textbf{Examples i medisin:}
    \begin{itemize}
        \item Modell trent på data fra USA brukes i Norge
        \item Ny MR-skanner gir andre bildekarakteristikker
        \item COVID endret innleggelsesmønstre
    \end{itemize}
    
    {\small
    \begin{alertblock}{Consequence}
        Modellen kan \textbf{\href{https://en.wikipedia.org/wiki/Silent_failure}{feile stille}} -- gir predictioner med høy konfidens that is feil!
    \end{alertblock}
    }
\end{frame}

\begin{frame}{T04: Explain the difference between epistemic and aleacountsic uncertainty}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{\href{https://en.wikipedia.org/wiki/Uncertainty_quantification\#Epistemic_uncertainty}{Epistemisk usikkerhet}:}
            \begin{itemize}
                \item Usikkerhet pga. \textbf{manglende kunnskap}
                \item ``Vi vet ikke nok ennå''
                \item \textbf{Kan reduseres} med mer data
                \item Modellen er usikker på områder med lite training data
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{Example:}
            \begin{itemize}
                \item Sjelden disease med få training examples
                \item Usikkerhet fordi modellen mangler erfaring
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{\href{https://en.wikipedia.org/wiki/Uncertainty_quantification\#Aleacountsic_uncertainty}{Aleacountsisk usikkerhet}:}
            \begin{itemize}
                \item \textbf{Iboende} tilfeldig variasjon i data
                \item ``Verden er usikker''
                \item \textbf{Kan ikke reduseres} med mer data
                \item Støy i goalinger, naturlig variasjon
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{Example:}
            \begin{itemize}
                \item To patienter med identiske features har ulike utfall
                \item Usikkerhet fordi utfall er genuint usikkert
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.5cm}
    \begin{block}{Practical significance}
        Epistemisk usikkerhet signaliserer når modellen bør ``si ifra'' at den er usikker, og mennesker bør overta.
    \end{block}
\end{frame}

\section{Human-Machine Interaction}

\begin{frame}{T05: Describe human-in-the-loop (HITL) systems}
    \textbf{\href{https://en.wikipedia.org/wiki/Human-in-the-loop}{Human-in-the-loop} (HITL):}
    \vspace{-0.1cm}
    \begin{itemize}
        \item Mennesker er \textbf{integrert} i AI-systemets beslutningsprosess
        \item AI gir anbefalinger, mennesker tar endelige beslutninger
    \end{itemize}
    
    \vspace{-0.05cm}
    \textbf{Tre hovedvarianter:}
    \vspace{-0.1cm}
    \begin{enumerate}
        \item \textbf{\href{https://en.wikipedia.org/wiki/Human-in-the-loop}{Human-in-the-loop}:} Menneske involveres i hver beslutning
        \item \textbf{\href{https://en.wikipedia.org/wiki/Human-on-the-loop}{Human-on-the-loop}:} Menneske overvåker og kan overstyre
        \item \textbf{\href{https://en.wikipedia.org/wiki/Automation\#Levels_of_automation}{Human-out-of-the-loop}:} Full autonomi (ikke anbefalt i medisin)
    \end{enumerate}
    
    \vspace{-0.05cm}
    \textbf{Advantages med HITL i medisin:}
    \vspace{-0.1cm}
    \begin{itemize}
        \item Kombinerer AI-effektivitet med menneskelig ekspertise
        \item Fanger opp AI-feil før de får konsekvenser
        \item Opprettholder clinical responsibility, bygger tillit gradvis
    \end{itemize}
    
    {\footnotesize
    \begin{alertblock}{\href{https://artificialintelligenceact.eu/}{EU AI Act}}
        Høyrisiko AI-systemer \textbf{krever} effektiv menneskelig tilsyn.
    \end{alertblock}
    }
\end{frame}

\begin{frame}{T06: Discuss the importance of continuous monicountsing}
    \textbf{Why \href{https://en.wikipedia.org/wiki/Continuous_monicountsing}{kontinuerlig monicountsering}?}
    \begin{itemize}
        \item ML-models \textbf{degraderer over tid} (model drift)
        \item Verden endrer seg, data endrer seg
        \item Feil i produksjon kan ha alvorlige konsekvenser
    \end{itemize}
    
    \vspace{0.2cm}
    \textbf{Hva bør overvåkes?}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Ytelsesmetrikker:}
            \begin{itemize}
                \item Accuracy, precision, recall over tid
                \item Kalibrering av konfidensscorer
                \item Sammenligning med baseline
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Datakarakteristikker:}
            \begin{itemize}
                \item Input-advantageing (drift-deteksjon)
                \item Andel out-of-distribution input
                \item Feature-statistikk
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.2cm}
    \textbf{Tiltak ved problemer:}
    \begin{itemize}
        \item \textbf{Re-training} med ferske data -- oppdater modellen på nyere, representative data
        \item \textbf{Varsling og escalesing} -- automatic varsling ved ytelsesfall or drift
        \item \textbf{Fallback} -- bytt til enklere modell or menneskelig vurdering ved usikkerhet
    \end{itemize}
\end{frame}

\section{Security Threats}

\begin{frame}{T07: Know about adversarial attacks}
    \textbf{\href{https://en.wikipedia.org/wiki/Adversarial_machine_learning}{Adversarial attacks} = bevisst manipulering av AI-input}
    
    \begin{itemize}
        \item Små, tilsynelatende usynlige endringer som lurer modellen
        \item Kan få en korrekt klassifikasjon til å bli \textbf{fullstendig feil}
    \end{itemize}
    
    \vspace{0.2cm}
    \textbf{Examples:}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Imager:}
            \begin{itemize}
                \item Piksel-perturbasjoner usynlige for mennesker
                \item ``Panda'' $\rightarrow$ ``Gibbon'' med 99\% konfidens
                \item \href{https://arxiv.org/abs/1707.08945}{Klistremerker som lurer selvkjørende biler}
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Tekst (LLM):}
            \begin{itemize}
                \item \href{https://en.wikipedia.org/wiki/Prompt_injection}{Prompt injection}
                \item \href{https://en.wikipedia.org/wiki/Jailbreaking_(LLM)}{Jailbreaking}
                \item Omgå \href{https://en.wikipedia.org/wiki/Content_moderation}{sikkerhetsfiltre}
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.15cm}
    {\small
    \begin{alertblock}{Medical risk og fotherwisevar}
        \textbf{Risiko:} En angriper kunne manipulere et røntgenbilde slik at AI overser patologi, or skaper falske funn. \\
        \textbf{Fotherwisevar:} Adversarial training, input-validation, ensemble-metoder, robustnessstesting.
    \end{alertblock}
    }
\end{frame}
