\section{Introduction to Generative AI}

\begin{frame}{G01: Define generative AI and distinguish from discriminative AI}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Discriminative AI:}
            \begin{itemize}
                \item Learns to \textbf{classify} or distinguish between categories
                \item Models $P(y|x)$ - probability of class $y$ given input $x$
                \item Examples: ``Is this cancer?'', ``Which disease?''
                \item CNN for image classification
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Generative AI:}
            \begin{itemize}
                \item Learns to \textbf{create} new content
                \item Models $P(x)$ or $P(x|y)$ - the data distribution itself
                \item Examples: Text, images, code, music
                \item LLMs, diffusion models
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{block}{\footnotesize Key difference}
        \footnotesize
        \textbf{Discriminative:} Input $x$ $\rightarrow$ Category/Label $y$ \quad | \quad
        \textbf{Generative:} Prompt $\rightarrow$ New content
    \end{block}

    \begin{alertblock}{\footnotesize Medical application}
        \footnotesize Generative AI: Summarize medical records, generate reports, answer questions, synthesize training data
    \end{alertblock}
\end{frame}

\section{Transformer Architecture}

\begin{frame}{G02: Explain the \href{https://en.wikipedia.org/wiki/Attention_(machine_learning)}{self-attention} mechanism at a conceptual level}
    \textbf{Self-attention: ``What is relevant for what?''}

    \vspace{0.2cm}
    \textbf{Intuition:}
    \begin{itemize}
        \item For each word: Look at \textbf{all} other words in the sentence
        \item Compute a \textbf{weighted sum} based on relevance
        \item Captures dependencies over long distances
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Example:}
    \begin{center}
        \textit{``The patient took the \textbf{medication}, and \textbf{it} helped with the pain.''}
    \end{center}
    \begin{itemize}
        \item Self-attention connects ``it'' to ``medication'' (not ``pain'')
    \end{itemize}

    \vspace{0.1cm}
    \begin{block}{\footnotesize \href{https://jalammar.github.io/illustrated-transformer/\#self-attention-at-a-high-level}{Query, Key, Value (QKV)}}
        \footnotesize Each token generates three vectors: \\
        \href{https://en.wikipedia.org/wiki/Attention_(machine_learning)\#Queries,_keys,_and_values}{\textbf{Query}} ``What am I looking for?'' |
        \href{https://en.wikipedia.org/wiki/Attention_(machine_learning)\#Queries,_keys,_and_values}{\textbf{Key}} ``What can I offer?'' |
        \href{https://en.wikipedia.org/wiki/Attention_(machine_learning)\#Queries,_keys,_and_values}{\textbf{Value}} ``What is my content?''
    \end{block}
\end{frame}

\begin{frame}{G03: Describe the transformer architecture (\href{https://en.wikipedia.org/wiki/Transformer_(deep_learning_architecture)\#Encoder}{encoder}-\href{https://en.wikipedia.org/wiki/Transformer_(deep_learning_architecture)\#Decoder}{decoder})}
    \textbf{Transformer Architecture (\href{https://arxiv.org/abs/1706.03762}{Vaswani et al., 2017}):}

    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Encoder:}
            \begin{itemize}
                \item Processes input sequence
                \item Builds context-rich representation
                \item Used in: \href{https://arxiv.org/abs/1810.04805}{BERT}, \href{https://en.wikipedia.org/wiki/Word_embedding}{embeddings}
            \end{itemize}

            \vspace{0.2cm}
            \textbf{Decoder:}
            \begin{itemize}
                \item Generates output sequentially
                \item Uses \href{https://en.wikipedia.org/wiki/Attention_(machine_learning)\#Causal_attention}{masked self-attention}
                \item Used in: GPT, Claude, Gemini
            \end{itemize}

            \vspace{0.2cm}
            \textbf{Full encoder-decoder:}
            \begin{itemize}
                \item Translation, summarization
                \item \href{https://arxiv.org/abs/1910.10683}{T5}, \href{https://arxiv.org/abs/2210.11416}{Flan-T5}
            \end{itemize}
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{Key components:}
            \begin{enumerate}
                \item \href{https://en.wikipedia.org/wiki/Attention_(machine_learning)\#Multi-head_attention}{Multi-head self-attention}
                \item \href{https://en.wikipedia.org/wiki/Feedforward_neural_network}{Feed-forward network}
                \item \href{https://en.wikipedia.org/wiki/Layer_normalization}{Layer normalization}
                \item \href{https://en.wikipedia.org/wiki/Residual_neural_network}{Residual connections}
                \item \href{https://en.wikipedia.org/wiki/Positional_encoding}{Positional encoding}
            \end{enumerate}

            \vspace{0.2cm}
            \begin{block}{\footnotesize GPT = ``Decoder-only''}
                \footnotesize Large language models (LLMs) typically use only the decoder part for text generation.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\section{LLM Fundamentals}

\begin{frame}{G04: Explain what \href{https://en.wikipedia.org/wiki/Lexical_analysis\#Token}{tokens} are and how \href{https://en.wikipedia.org/wiki/Tokenization_(lexical_analysis)}{tokenization} works}
    \textbf{Tokens = the ``atoms'' of text}
    \begin{itemize}
        \item LLMs don't read letters or words - they read \textbf{tokens}
        \item Token $\approx$ subword, about 4 characters per token (for English)
    \end{itemize}

    \vspace{0.15cm}
    \textbf{Tokenization - example:}
    \begin{center}
        \footnotesize
        \texttt{``The patient has diabetes''} $\rightarrow$ \texttt{[``The'', `` patient'', `` has'', `` diab'', ``etes'']}
    \end{center}

    \vspace{0.1cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{\href{https://en.wikipedia.org/wiki/Byte_pair_encoding}{BPE} and language differences:}
            \footnotesize
            \begin{itemize}
                \item Builds vocabulary iteratively
                \item GPT-4: approx. 100,000 tokens
                \item \textbf{Chinese:} Each character $\approx$ 1--2 tokens
                \item Simplified/traditional: Different tokens
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Important to know:}
            \footnotesize
            \begin{itemize}
                \item Long/rare words = more tokens
                \item Affects cost and speed
                \item \href{https://github.com/openai/tiktoken}{\texttt{tiktoken}} (OpenAI) for counting
                \item Different models = different tokenizers
            \end{itemize}
        \end{column}
    \end{columns}

    \begin{block}{\footnotesize Why tokens?}
        \footnotesize More efficient than letters (too short) or whole words (vocabulary too large).
    \end{block}
\end{frame}

\begin{frame}{G05: Understand the concept of \href{https://en.wikipedia.org/wiki/Context_window}{context window}}
    \textbf{Context window = the model's ``working memory''}
    \begin{itemize}
        \item Maximum number of tokens the model can ``see'' at once
        \item Includes both \textbf{input} (prompt) and \textbf{output} (response)
    \end{itemize}

    \vspace{0.2cm}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Example context windows:}
            \footnotesize
            \begin{tabular}{lrl}
                \toprule
                Model & Context & $\approx$ Pages \\
                \midrule
                GPT-4o & 128K & $\sim$150 pages \\
                GPT-5 & 256K+ & $\sim$300 pages \\
                Claude Opus 4.5 & 200K & $\sim$250 pages \\
                Gemini 3 Pro & 2M+ & Entire books! \\
                \bottomrule
            \end{tabular}
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{Input vs. output:}
            \footnotesize
            \begin{itemize}
                \item \textbf{Input context:} How much the model can read
                \item \textbf{Output context:} How long a response it can give
                \item Output often smaller (4K--32K tokens)
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.1cm}
    \begin{alertblock}{\footnotesize Limitations}
        \footnotesize
        Long texts must be truncated. The model ``forgets'' outside the window. Larger window = more expensive/slower.
    \end{alertblock}
\end{frame}

\begin{frame}{G06: Explain what \href{https://en.wikipedia.org/wiki/Softmax_function\#Softmax_temperature}{temperature} means in text generation}
    \textbf{Temperature = control over ``creativity''}

    \vspace{0.15cm}
    \textbf{Technical:}
    \begin{itemize}
        \item Scales logits before softmax: $p_i = \frac{e^{z_i/T}}{\sum_j e^{z_j/T}}$
        \item \textbf{Effect:} Low $T$ $\rightarrow$ sharper distribution (highest logit dominates); High $T$ $\rightarrow$ flatter distribution (more randomness)
    \end{itemize}

    \vspace{0.2cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Low temperature (0.0--0.3):}
            \begin{itemize}
                \item More deterministic
                \item Chooses most likely tokens
                \item Consistent, ``safe''
                \item \faCheck~Medical info, facts
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{High temperature (0.7--1.0+):}
            \begin{itemize}
                \item More random/creative
                \item More surprising choices
                \item More variation
                \item \faCheck~Creative writing, brainstorming
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.15cm}
    \begin{block}{\footnotesize Recommendation for medicine}
        \footnotesize Use \textbf{low temperature} (0.0--0.3) for fact-based tasks. Higher temperature increases hallucination risk.
    \end{block}
\end{frame}

\section{Prompt Engineering}

\begin{frame}{G07: Apply \href{https://en.wikipedia.org/wiki/Zero-shot_learning}{zero-shot} prompting}
    \textbf{Zero-shot = no examples} - ask the model to solve the task \textbf{without showing examples}

    \vspace{0.1cm}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Examples of zero-shot prompts:}
            \begin{block}{\footnotesize Zero-shot prompts}
                \scriptsize
                \begin{enumerate}
                    \item \texttt{Classify symptom as acute/chronic: ``Headache for 3 months''}
                    \item \texttt{Summarize the medical note in 3 sentences.}
                    \item \texttt{Translate to patient-friendly language: ``Bilateral pneumonia''}
                    \item \texttt{Extract all medications from this text.}
                    \item \texttt{Is this lab value abnormal? Hb 8.2 g/dL}
                \end{enumerate}
            \end{block}
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{When does zero-shot work?}
            \footnotesize
            \begin{itemize}
                \item The model has seen similar tasks
                \item The task is clearly defined
            \end{itemize}

            \vspace{0.1cm}
            \begin{alertblock}{\footnotesize Limitations}
                \scriptsize
                For specialized tasks, zero-shot may give poor results. Then \textbf{few-shot} is needed.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{G08: Apply \href{https://en.wikipedia.org/wiki/Few-shot_learning}{few-shot prompting}}
    \textbf{Few-shot = some examples}
    
    \begin{itemize}
        \item Give the model \textbf{2--5 examples} of desired input-output
        \item The model learns the pattern ``in-context''
    \end{itemize}
    
    \vspace{0.3cm}
    \begin{block}{Few-shot prompt}
        \small
        \texttt{Classify symptom duration:} \\
        \texttt{Symptom: ``Chest pain for 30 minutes'' → Acute} \\
        \texttt{Symptom: ``Back pain for 2 years'' → Chronic} \\
        \texttt{Symptom: ``Fever since yesterday'' → Acute} \\
        \texttt{---} \\
        \texttt{Symptom: ``Headache for 3 months'' →}
    \end{block}
    
    \vspace{0.3cm}
    \textbf{Tips for effective few-shot:}
    \begin{itemize}
        \item Choose \textbf{representative} examples
        \item Show the \textbf{variation} in possible outputs
        \item Keep examples \textbf{consistent} in format
        \item Start with 3--5 examples, adjust as needed
    \end{itemize}
\end{frame}

\begin{frame}{G09: Apply \href{https://en.wikipedia.org/wiki/Prompt_engineering\#Chain-of-thought}{Chain-of-Thought (CoT)} prompting}
    \textbf{Chain-of-Thought = show the reasoning}
    \begin{itemize}
        \item Ask the model to \textbf{think step-by-step}
        \item Improves performance on complex reasoning tasks
    \end{itemize}

    \vspace{0.2cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{\footnotesize Without CoT}
                \footnotesize
                \texttt{Should the patient be referred?} \\
                \texttt{→ ``Yes''}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{\footnotesize With CoT}
                \footnotesize
                \texttt{Think step-by-step:} \\
                \texttt{1. Symptoms: X, Y, Z} \\
                \texttt{2. Severity: Moderate} \\
                \texttt{→ Conclusion: Should be referred}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.15cm}
    \textbf{Variants:}
    \begin{itemize}
        \item \textbf{Zero-shot CoT:} ``Let's think step by step''
        \item \textbf{Few-shot CoT:} Show examples with reasoning
    \end{itemize}

    \begin{block}{\footnotesize Built-in CoT in newer LLMs}
        \footnotesize ChatGPT o1/o3 (Thinking), Claude Opus 4.5 (Extended Thinking), Gemini 3 (Deep Think) have CoT built-in.
    \end{block}
\end{frame}

\begin{frame}{G10: Describe best practices for \href{https://en.wikipedia.org/wiki/Prompt_engineering\#System_prompts}{system prompts}}
    \textbf{System prompt = instruction that sets context}

    \vspace{0.2cm}
    \textbf{Elements of a good system prompt:}
    \begin{enumerate}
        \item \textbf{Role:} ``You are a medical assistant...''
        \item \textbf{Context:} ``The user is a healthcare professional...''
        \item \textbf{Behavior:} ``Answer concisely and precisely...''
        \item \textbf{Limitations:} ``Do not give diagnoses...''
        \item \textbf{Format:} ``Answer in English, use bullet points...''
    \end{enumerate}

    \vspace{0.15cm}
    \begin{block}{\footnotesize Example system prompt}
        \scriptsize
        \texttt{You are a medical AI assistant for healthcare professionals. Be precise and evidence-based. Reference sources when possible. Never give definitive diagnoses - suggest differential diagnoses. If uncertain, say so.}
    \end{block}

    \begin{alertblock}{\footnotesize Tip}
        \footnotesize Test and iterate on system prompts! Small changes can have large effects.
    \end{alertblock}
\end{frame}

\begin{frame}{G18: Handle uncertainty and hallucination risk}
    \textbf{Prompt patterns for medical uncertainty handling:}
    
    \vspace{0.2cm}
    \begin{block}{\footnotesize Anti-hallucination prompt (GPT-5.2 style)}
        \scriptsize
        \texttt{<uncertainty\_and\_ambiguity>} \\
        \texttt{- For ambiguous symptoms: Present differential diagnoses with probability} \\
        \texttt{- Clearly state knowledge limitations and update date} \\
        \texttt{- Always refer to current guidelines and clinical judgment} \\
        \texttt{- When uncertain: Escalate to higher urgency level} \\
        \texttt{</uncertainty\_and\_ambiguity>}
    \end{block}
    
    \vspace{0.15cm}
    \textbf{Self-check for high-risk tasks:}
    \footnotesize
    \begin{enumerate}
        \item Scan own response for unstated assumptions
        \item Verify that numbers and claims are grounded in context
        \item Avoid overly strong language (``always'', ``guaranteed'')
        \item Qualify conclusions with uncertainty level
    \end{enumerate}
    
    \normalsize
    \begin{alertblock}{\footnotesize Rule of thumb}
        \footnotesize AI = decision support, \textbf{not} decision maker. Always clinical validation!
    \end{alertblock}
\end{frame}

\section{Challenges with LLM}

\begin{frame}{G11: Define \href{https://en.wikipedia.org/wiki/Hallucination_(artificial_intelligence)}{hallucination} and its implications}
    \textbf{Hallucination = the model ``makes up'' incorrect information}
    \begin{itemize}
        \item LLMs generate text that \textit{sounds} correct but is \textit{actually wrong}
        \item Also called \href{https://en.wikipedia.org/wiki/Confabulation_(neural_networks)}{\textit{confabulation}} - both terms are used in the literature
    \end{itemize}

    \vspace{0.15cm}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Types of hallucinations:}
            \footnotesize
            \begin{itemize}
                \item \textbf{Factual errors:} Wrong statistics, dates, names
                \item \textbf{Fabricated sources:} References that don't exist
                \item \textbf{Logical errors:} Inconsistent reasoning
            \end{itemize}
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{Development:}
            \footnotesize
            \begin{itemize}
                \item Newer models hallucinate \textit{less}
                \item Cannot be \textit{completely} eliminated
                \item RAG and CoT reduce risk
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.1cm}
    \begin{alertblock}{\footnotesize Implications in medicine}
        \footnotesize
        Wrong dosage can be life-threatening. \textbf{Always verify} against reliable sources! LLM = decision support, \textbf{not} replacement for professional expertise.
    \end{alertblock}
\end{frame}

\begin{frame}{G13: Explain the concept of \href{https://en.wikipedia.org/wiki/Foundation_model}{foundation model}}
    \textbf{Foundation model = pretrained on enormous amounts of data}
    \begin{itemize}
        \item Trained on terabytes of text/images without specific task
        \item Can be adapted (\textit{finetuned}) to many different tasks
        \item Costs millions to train from scratch
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Characteristics:}
    \begin{enumerate}
        \item \textbf{Large scale:} Billions of parameters
        \item \textbf{Self-supervised learning:} No labels necessary
        \item \textbf{Emergent capabilities:} Abilities that ``emerge'' at scale
        \item \textbf{Adaptable:} Finetuning, prompting, RAG
    \end{enumerate}

    \vspace{0.15cm}
    \begin{block}{\footnotesize Examples of foundation models (2025)}
        \scriptsize
        \textbf{Text:} \href{https://openai.com/gpt-5}{GPT-5}, \href{https://www.anthropic.com/claude}{Claude 4}, \href{https://llama.meta.com/}{Llama 4} |
        \textbf{Images:} \href{https://openai.com/dall-e-3}{DALL-E 3}, \href{https://stability.ai/}{Stable Diffusion 3} |
        \textbf{Multimodal:} \href{https://deepmind.google/gemini}{Gemini 3}, \href{https://openai.com/sora}{Sora}
    \end{block}
\end{frame}

\begin{frame}{G14: Describe \href{https://en.wikipedia.org/wiki/Retrieval-augmented_generation}{RAG} (Retrieval-Augmented Generation)}
    \textbf{RAG = connect LLM to external knowledge base}

    \vspace{0.2cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{The problem RAG solves:}
            \begin{itemize}
                \item LLM has ``frozen'' knowledge
                \item Can hallucinate facts
                \item Lacks domain-specific info
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{How RAG works:}
            \begin{enumerate}
                \item \textbf{Query:} User asks question
                \item \textbf{Retrieval:} Fetch documents
                \item \textbf{Augmentation:} Add to prompt
                \item \textbf{Generation:} LLM responds
            \end{enumerate}
        \end{column}
    \end{columns}

    \vspace{0.15cm}
    \begin{block}{\footnotesize Medical application}
        \scriptsize
        \begin{itemize}
            \item Connect LLM to guidelines, drug databases, \href{https://www.uptodate.com/}{UpToDate}
            \item Answer based on patient record (with permissions)
            \item Reduces hallucination by grounding answers in sources
            \item Clinical decision support with updated evidence
            \item Search institution-specific procedures and protocols
        \end{itemize}
    \end{block}
\end{frame}
