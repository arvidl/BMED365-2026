\begin{frame}{M01: Define \href{https://en.wikipedia.org/wiki/Machine_learning}{machine learning} og distinguish from tradisjonell programmering}

    \begin{columns}
        \begin{column}{0.48\textwidth}
            \textbf{Tradisjonell programmering:}
            \begin{itemize}
                \item Eksplisitte regler definert av programmerer
                \item \texttt{if-else} logikk, deterministisk
                \item Input + Regler $\rightarrow$ Output
                \item Krever full domenekunnskap på forhånd
            \end{itemize}

            \vspace{2mm}
            \textbf{Machine Learning:}
            \begin{itemize}
                \item Lærer regler fra data automatic
                \item Mønstergjenkjenning, statistisk
                \item Input + Output $\rightarrow$ Regler (modell)
                \item Skalerer med datamengde
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{\scriptsize \href{https://www.cs.cmu.edu/~tom/mlbook.html}{Definition (Tom Mitchell, 1997)}}
                \scriptsize
                \textit{``A computer program is said to \textbf{\textit{learn from experience}} E with respect to some task T and some performance measure P, if its performance on T, as measured by P, improves with experience E.''}
            \end{block}

            \begin{block}{\scriptsize Definition (Tom Mitchell, 1997) -- på notherwisek}
                \scriptsize
                \textit{``Et dataprogram sies å \textbf{\textit{lære av erfaring}} E med hensyn til en task T og et ytelsesgoal P, dersom dets ytelse på T, goalt ved P, forbedres med erfaring E.''}
            \end{block}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{M02: \href{https://en.wikipedia.org/wiki/Supervised_learning}{Supervised} vs. \href{https://en.wikipedia.org/wiki/Unsupervised_learning}{Unsupervised} læring}

    \begin{columns}
        \begin{column}{0.48\textwidth}
            \textbf{\faCheckCircle\ Supervised (veiledet) læring:}
            \begin{itemize}
                \item Data har \textbf{labels} (fasit/merkelapper)
                \item Modellen lærer input$\rightarrow$output-mapping
                \item Typer: Classification, regression
                \item Medical: Diagnostisering, prognosis
                \item Krever manual annotering av data
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{\faSearch\ Unsupervised (ikke-veiledet) læring:}
            \begin{itemize}
                \item Data \textbf{uten} labels
                \item Finner skjulte mønstre/strukturer
                \item Typer: Klynging, dimensjonsreduksjon
                \item Medical: Pasient-subgrupper
                \item Ingen fasit -- utfotherwisekende analyse
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{2mm}
    \begin{block}{\footnotesize \href{https://en.wikipedia.org/wiki/Semi-supervised_learning}{Semi-supervised} (delvis veiledet) -- prinsipper og applylser}
        \scriptsize
        Kombinerer få merkede eksempler med mange umerkede. Nyttig når annotering er dyrt/tidkrevende.\\[2pt]
        \textbf{Prinsipper:}
        \textit{Smoothness} -- nærliggende datapunkter har ofte samme label. \textit{Tabulært:} patienter med like lab-valueer har ofte samme diagnosis. \textit{Image:} MR-snitt med lignende pikselintensiteter viser lignende patologi.
        \textit{Cluster} -- data grupperer seg naturlig; punkter i samme klynge tilhører ofte samme klasse. \textit{Tabulært:} patienter med diabetes klynger seg i feature-rommet. \textit{Image:} hjerne-MR av friske vs. Alzheimer danner separate klynger.
        \href{https://en.wikipedia.org/wiki/Manifold_hypothesis}{\textit{Manifold}} -- et 256$\times$256 bilde = 65536 piksler, men meningsfull variasjon bestemmes av få fakcountser (anatomi, patologi). Imagene ``fyller'' ikke hele rommet, men ligger på en lavdim. flate.\\[2pt]
        \textbf{Applications:} Medical bildesegmentation, tekstclassification, talegjenkjenning, drugoppdagelse, genomikk.
    \end{block}

\end{frame}

                \end{enumerate}
            \end{block}

\begin{frame}{M06: \href{https://en.wikipedia.org/wiki/Bias\%E2\%80\%93variance_tradeoff}{Bias-Variance Trade-off}}

    \textbf{Totalt predictionsfeil:}
    \[ \text{MSE} = \underbrace{\text{Bias}^2}_{\text{systematisk feil}} + \underbrace{\text{Varians}}_{\text{tilfeldig feil}} + \underbrace{\sigma^2}_{\text{irreducible}} \]

    \vspace{1mm}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \textbf{Bias (systematisk feil):}
            \begin{itemize}
                \item Feil pga. forenklet modell
                \item Høy bias $\rightarrow$ \textbf{underfitting}
                \item Modellen ``misser'' mønsteret
            \end{itemize}

            \vspace{2mm}
            \textbf{Varians (tilfeldig feil):}
            \begin{itemize}
                \item Sensitiv for små endringer i data
                \item Høy varians $\rightarrow$ \textbf{overfitting}
                \item Modellen ``husker'' støy
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{center}
                \begin{tikzpicture}[scale=0.65]
                    \draw[->] (0,0) -- (5,0) node[below] {\small Kompleksitet};
                    \draw[->] (0,0) -- (0,4) node[left] {\small Feil};

                    \draw[uibblue, thick] (0.5,3) .. controls (2,1) and (3,0.5) .. (4.5,0.3);
                    \draw[uibred, thick] (0.5,0.3) .. controls (2,0.5) and (3,1.5) .. (4.5,3.5);
                    \draw[black, thick, dashed] (0.5,3.3) .. controls (1.5,1.5) and (2.5,1.2) .. (2.5,1.2) .. controls (3,1.5) and (4,2.5) .. (4.5,3.8);

                    \node[uibblue] at (1, 3.5) {\tiny Bias²};
                    \node[uibred] at (4, 3.5) {\tiny Varians};
                    \node at (2.5, 0.7) {\tiny Optimal};

                    \draw[dotted] (2.5, 0) -- (2.5, 1.2);
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}

    \begin{block}{\scriptsize The goal}
        \scriptsize Finn balansen som minimerer \textbf{total feil} på nye data. $\sigma^2$ = støy i data (kan ikke reduseres).
    \end{block}

\end{frame}

\begin{frame}{M07: \href{https://en.wikipedia.org/wiki/Cross-validation_(statistics)\#k-fold_cross-validation}{K-fold cross-validation}}
    
    \textbf{Problem:} Ett simplet train/test-split kan gi tilfeldige resulter
    
    \vspace{3mm}
    
    \textbf{Solution: K-fold Cross-Validation}
    \begin{enumerate}
        \item Del data i $k$ like scountse deler (``folds'')
        \item For hver fold: bruk den som test, resten som training
        \item Average av alle $k$ evaluationer
    \end{enumerate}
    
    \vspace{3mm}
    
    \begin{center}
        \begin{tikzpicture}[scale=0.8]

            \fill[uibred!60] (0,2) rectangle (1,2.4);

\begin{frame}{M08: \href{https://en.wikipedia.org/wiki/Baseline_(machine_learning)}{Baseline model}}

    \begin{block}{\footnotesize What is a baseline?}
        \footnotesize En \textbf{simple referansemodell} som vår ML-modell må slå to være nyttig.
    \end{block}

    \vspace{2mm}
    \textbf{Examples på baselines:}

    \begin{itemize}
        \item \textbf{Classification:} Prediker alltid majoritetsklassen
        \begin{itemize}
            \item Dataset med 90\% friske, 10\% syke $\rightarrow$ baseline accuracy = 90\%
        \end{itemize}
        \item \textbf{Regression:} Prediker alltid gjennomsnittsvalueen
        \item \textbf{Tidsserie:} Bruk forrige value som prediction
    \end{itemize}

    \vspace{2mm}
    \begin{alertblock}{\footnotesize Why important?}
        \footnotesize
        90\% accuracy høres imponerende ut, men ikke hvis baseline er 90\%! Viser om ML tilfører \textbf{faktisk value}. Avslører ubalanserte dataset.
    \end{alertblock}

\end{frame}

        \end{column}
